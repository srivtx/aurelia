   415→/* bidirectional pairing: A pairs with B ⇒ B pairs with A */
   416→export function getMatchesFor(id: string): { color: WardrobeColor; why: string }[] {
   417→  const out: { color: WardrobeColor; why: string }[] = [];
   418→  for (const c of wardrobeColors) {
   419→    if (c.id === id) continue;
   420→    const direct = c.pairs.find((p) => p.id === id);
   421→    if (direct) out.push({ color: c, why: direct.why });
   422→  }
   423→  const self = colorById(id);
   424→  if (self) for (const p of self.pairs) {
   425→    const pc = colorById(p.id);
   426→    if (pc && !out.find((o) => o.color.id === pc.id)) out.push({ color: pc, why: p.why });
   427→  }
   428→  return out;
   429→}
